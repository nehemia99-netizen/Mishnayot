לפני לימוד משנה›
לְשֵׁם יִחוּד קֻדְשָׁא בְּרִיךְ הוּא וּשְׁכִינְתֵּיהּ, הֲרֵינִי בָּא לִלְמֹד מִשְׁנָה לִשְׁמָהּ.

יְהִי רָצוֹן מִלְּפָנֶיךָ ה' אֱלֹהַי וֵאלֹהֵי אֲבוֹתַי, שֶׁתָּכִין לְבָבִי וְתָאִיר עֵינַי בְּתַלְמוּד תּוֹרָתֶךָ, וְתִתֶּן בְּלִבִּי בִּינָה לְהָבִין וּלְהַשְׂכִּיל, לִשְׁמֹעַ לִלְמֹד וּלְלַמֵּד, לִשְׁמֹר וְלַעֲשׂוֹת. וְיִהְיוּ דִּבְרֵי הַמִּשְׁנָה שֶׁאֲנִי לוֹמֵד חֲשׁוּבִים וּמְקֻבָּלִים לְפָנֶיךָ, וְזַכֵּנִי שֶׁתְּהֵא תּוֹרָתְךָ אֻמָּנוּתִי. אָמֵן.
אחרי לימוד משנה›
רַבִּי חֲנַנְיָא בֶּן עֲקַשְׁיָא אוֹמֵר: רָצָה הַקָּדוֹשׁ בָּרוּךְ הוּא לְזַכּוֹת אֶת יִשְׂרָאֵל, לְפִיכָךְ הִרְבָּה לָהֶם תּוֹרָה וּמִצְוֹת, שֶׁנֶּאֱמַר: ה' חָפֵץ לְמַעַן צִדְקוֹ יַגְדִּיל תּוֹרָה וְיַאְדִּיר.

יהי רצון (אחרי לימוד משנה)›
יְהִי רָצוֹן מִלְּפָנֶיךָ ה' אֱלֹהַי וֵאלֹהֵי אֲבוֹתַי, שֶׁתְּהֵא הַמִּשְׁנָה הַזֹּאת שֶׁלָּמַדְתִּי חֲשׁוּבָה וּמְקֻבֶּלֶת לְפָנֶיךָ, וְתַעֲלֶה לְךָ לְרָצוֹן כְּקָרְבָּן, וּזְכוּת הַתַּנָּאִים הַקְּדוֹשִׁים יַעֲמֹד לִי וּלְכָל יִשְׂרָאֵל.

וְזַכֵּנִי לִלְמֹד וּלְלַמֵּד, לִשְׁמֹר וְלַעֲשׂוֹת וּלְקַיֵּם אֶת כָּל דִּבְרֵי תַלְמוּד תּוֹרָתֶךָ בְּאַהֲבָה. וְהָאֵר עֵינַי בְּתוֹרָתֶךָ, וְדַבֵּק לִבִּי בְּמִצְוֹתֶיךָ, וְיַחֵד לְבָבִי לְאַהֲבָה וּלְיִרְאָה אֶת שְׁמֶךָ. אָמֵן.
🕯️ יהי רצון לעילוי נשמה›
יְהִי רָצוֹן מִלְּפָנֶיךָ, אֲדֹנָי אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ, שֶׁיְּהֵא לִימוּד מִשְׁנָיוֹת אֵלּוּ (וּתְפִלָּתֵנוּ) לְעִלּוּי נִשְׁמַת רוּחַ וְנֶפֶשׁ (פלוני בן פלונית). וּבִזְכוּת זֶה תִּפָּתַחְנָה לָהּ שַׁעֲרֵי רַחֲמִים וְרָצוֹן, וְשַׁעֲרֵי גַן עֵדֶן, וּתְקַבֵּל אוֹתָהּ בְּאַהֲבָה וּבְחֶמְלָה. וְתִשְׁכּוֹן בִּצְרוֹר הַחַיִּים אֶת נִשְׁמַת אַבְרָהָם יִצְחָק וְיַעֲקֹב, שָׂרָה רִבְקָה רָחֵל וְלֵאָה, וְעִם שְׁאָר צַדִּיקִים וְצִדְקָנִיּוֹת שֶׁבְּגַן עֵדֶן, וְאָמֵן.

מי שברך לחולים›
מִי שֶׁבֵּרַךְ אֲבוֹתֵינוּ אַבְרָהָם יִצְחָק וְיַעֲקֹב, מֹשֶׁה אַהֲרֹן דָּוִד וּשְׁלֹמֹה, הוּא יְבָרֵךְ וִירַפֵּא אֶת הַחוֹלֶה [שם החולה] בֶּן/בַּת [שם האם] בַּעֲבוּר שֶׁ[שם המתפלל] יִתֵּן צְדָקָה בַּעֲבוּרוֹ/בַּעֲבוּרָהּ. בִּשְׂכַר זֶה, הַקָּדוֹשׁ בָּרוּךְ הוּא יִמָּלֵא רַחֲמִים עָלָיו/עָלֶיהָ לְהַחֲלִימוֹ וּלְרַפְּאוֹתוֹ, לְחַזְּקוֹ וּלְהַחֲיוֹתוֹ, וְיִשְׁלַח לוֹ/לָהּ מְהֵרָה רְפוּאָה שְׁלֵמָה מִן הַשָּׁמַיִם — רְפוּאַת הַנֶּפֶשׁ וּרְפוּאַת הַגּוּף — בְּתוֹךְ שְׁאָר חוֹלֵי יִשְׂרָאֵל, הַשְׁתָּא בַּעֲגָלָא וּבִזְמַן קָרִיב, וְאִמְרוּ: אָמֵן.
מי שברך לחיילים›
מִי שֶׁבֵּרַךְ אֲבוֹתֵינוּ אַבְרָהָם יִצְחָק וְיַעֲקֹב, הוּא יְבָרֵךְ אֶת חַיָּלֵי צְבָא הַהֲגַנָּה לְיִשְׂרָאֵל הָעוֹמְדִים עַל מִשְׁמַר אַרְצֵנוּ וְעָרֵי אֱלֹהֵינוּ, מִגְּבוּל הַלְּבָנוֹן וְעַד מִדְבַּר מִצְרַיִם, וּמִן הַיָּם הַגָּדוֹל עַד לְבוֹא הָעֲרָבָה, בַּיַּבָּשָׁה בָּאֲוִיר וּבַיָּם. יִתֵּן ה' אֶת אוֹיְבֵינוּ הַקָּמִים עָלֵינוּ נִגָּפִים לִפְנֵיהֶם. הַקָּדוֹשׁ בָּרוּךְ הוּא יִשְׁמֹר וְיַצִּיל אֶת חַיָּלֵינוּ מִכָּל צָרָה וְצוּקָה וּמִכָּל נֶגַע וּמַחֲלָה, וְיִשְׁלַח בְּרָכָה וְהַצְלָחָה בְּכָל מַעֲשֵׂה יְדֵיהֶם. יַדְבֵּר שׂוֹנְאֵינוּ תַּחְתֵּיהֶם, וִיעַטְּרֵם בְּכֶתֶר יְשׁוּעָה וּבַעֲטֶרֶת נִצָּחוֹן. וְנֹאמַר: אָמֵן.
תפילת שמירת הלשון — החפץ חיים›
רִבּוֹנוֹ שֶׁל עוֹלָם, יְהִי רָצוֹן מִלְּפָנֶיךָ, אֵל רַחוּם וְחַנּוּן, שֶׁתְּזַכֵּנִי הַיּוֹם וּבְכָל יוֹם לִשְׁמוֹר פִּי וּלְשׁוֹנִי מִלָּשׁוֹן-הָרָע וּרְכִילוּת וּמִקַּבָּלָתָם.

וְאֶזָּהֵר מִלְדַבֵּר אֲפִלּוּ עַל אִישׁ יְחִידִי, וְכָל שֶׁכֵּן מִלְדַבֵּר עַל כְּלַל יִשְׂרָאֵל אוֹ עַל חֵלֶק מֵהֶם. וְכָל שֶׁכֵּן מִלְּהִתְרַעֵם עַל מִדּוֹתָיו שֶׁל הַקָּדוֹשׁ-בָּרוּךְ-הוּא.

וְאֶזָּהֵר מִלְּדַבֵּר דִּבְרֵי שֶׁקֶר, חֲנֻפָּה, לֵצָנוּת, מַחֲלֹקֶת, כַּעַס, גַּאֲוָה, אוֹנָאַת-דְּבָרִים, הַלְבָּנַת-פָּנִים, וְכָל דְּבָרִים אֲסוּרִים.

וְזַכֵּנִי שֶׁלֹּא לְדַבֵּר כִּי-אִם דָּבָר הַצָּרִיךְ לְעִנְיָנֵי גוּפִי אוֹ נַפְשִׁי. וְשֶׁיִּהְיוּ כָּל מַעֲשַׂי לְשֵׁם שָׁמַיִם.

— מרן החפץ חיים, רבי ישראל מאיר הכהן מראדין זצ״ל
ברכה מעין שלוש›
ברכה אחרונה על מזונות · יין · פירות שבעת המינים
ניתן לבחור כמה אופציות יחד
איגרת הרמב״ן›
שְׁמַע בְּנִי מוּסַר אָבִיךָ וְאַל תִּטֹּשׁ תּוֹרַת אִמֶּךָ. תִּתְנַהֵג תָּמִיד לְדַבֵּר כָּל דְּבָרֶיךָ בְּנַחַת, לְכָל אָדָם וּבְכָל עֵת, וּבָזֶה תִּנָּצֵל מִן הַכַּעַס, שֶׁהִיא מִדָּה רָעָה לְהַחֲטִיא בְּנֵי אָדָם. וְכֵן אָמְרוּ רַזַ"ל (נדרים כב.): כָּל הַכּוֹעֵס — כָּל מִינֵי גֵּיהִנֹּם שׁוֹלְטִים בּוֹ, שֶׁנֶּאֱמַר (קהלת יא, י): "וְהָסֵר כַּעַס מִלִּבֶּךָ וְהַעֲבֵר רָעָה מִבְּשָׂרֶךָ", וְאֵין "רָעָה" אֶלָּא גֵּיהִנֹּם, שֶׁנֶּאֱמַר (משלי טז, ד): "וְגַם רָשָׁע לְיוֹם רָעָה". וְכַאֲשֶׁר תִּנָּצֵל מִן הַכַּעַס תַּעֲלֶה עַל לִבְּךָ מִדַּת הָעֲנָוָה, שֶׁהִיא מִדָּה טוֹבָה מִכָּל הַמִּדּוֹת הַטּוֹבוֹת, שֶׁנֶּאֱמַר (משלי כב, ד): "עֵקֶב עֲנָוָה יִרְאַת ה'".

וּבַעֲבוּר הָעֲנָוָה תַּעֲלֶה עַל לִבְּךָ מִדַּת הַיִּרְאָה, כִּי תִּתֵּן אֶל לִבְּךָ תָּמִיד: מֵאַיִן בָּאתָ וּלְאָן אַתָּה הוֹלֵךְ, וְשֶׁאַתָּה רִמָּה וְתוֹלֵעָה בְּחַיֶּיךָ וְאַף כִּי בְּמוֹתֶךָ, וְלִפְנֵי מִי אַתָּה עָתִיד לִתֵּן דִּין וְחֶשְׁבּוֹן — לִפְנֵי מֶלֶךְ הַכָּבוֹד, שֶׁנֶּאֱמַר (מלכים א ח, כז): "הִנֵּה הַשָּׁמַיִם וּשְׁמֵי הַשָּׁמַיִם לֹא יְכַלְכְּלוּךָ" — אַף כִּי לִבּוֹת בְּנֵי אָדָם, וְנֶאֱמַר (ירמיהו כג, כד): "הֲלוֹא אֶת הַשָּׁמַיִם וְאֶת הָאָרֶץ אֲנִי מָלֵא נְאֻם ה'", וְכַאֲשֶׁר תַּחְשׁוֹב אֶת כָּל אֵלֶּה — תִּירָא מִבּוֹרְאֶךָ וְתִשָּׁמֵר מִן הַחֵטְא, וּבַמִּדּוֹת הָאֵלֶּה תִּהְיֶה שָׂמֵחַ בְּחֶלְקֶךָ.

וְכַאֲשֶׁר תִּתְנַהֵג בְּמִדַּת הָעֲנָוָה לְהִתְבּוֹשֵׁשׁ מִכָּל אָדָם וּלְהִתְפַּחֵד מִמֶּנּוּ וּמִן הַחֵטְא — אֲזַי תִּשְׁרֶה עָלֶיךָ רוּחַ הַשְּׁכִינָה וְזִיו כְּבוֹדָהּ, וְחַיֵּי עוֹלָם הַבָּא.

וְעַתָּה בְּנִי, דַּע וּרְאֵה כִּי הַמִּתְגָּאֶה בְלִבּוֹ עַל הַבְּרִיּוֹת — מוֹרֵד הוּא בְּמַלְכוּת שָׁמַיִם, כִּי מִתְפָּאֵר הוּא בִּלְבוּשׁ מַלְכוּת שָׁמַיִם, שֶׁנֶּאֱמַר (תהלים צג, א): "ה' מָלָךְ גֵּאוּת לָבֵשׁ". וּבַמֶּה יִתְגָּאֶה הָאָדָם — אִם בְּעוֹשֶׁר, "ה' מוֹרִישׁ וּמַעֲשִׁיר" (שמואל א ב, ז); וְאִם בְּכָבוֹד — הֲלוֹא לֵאלֹהִים הוּא, שֶׁנֶּאֱמַר (דברי הימים א כט, יב): "וְהָעוֹשֶׁר וְהַכָּבוֹד מִלְּפָנֶיךָ"? וְאֵיךְ מִתְפָּאֵר בִּכְבוֹד קוֹנוֹ! וְאִם מִתְפָּאֵר בְּחָכְמָה — "מֵסִיר שָׂפָה לְנֶאֱמָנִים וְטַעַם זְקֵנִים יִקָּח" (איוב יב, כ); נִמְצָא הַכֹּל שָׁוֶה לִפְנֵי הַמָּקוֹם, כִּי בְּאַפּוֹ מַשְׁפִּיל גֵּאִים וּבִרְצוֹנוֹ מַגְבִּיהַּ שְׁפָלִים. לָכֵן הַשְׁפֵּל עַצְמְךָ וִינַשֵּׂאֲךָ הַמָּקוֹם.

עַל כֵּן אֲפָרֵשׁ לְךָ אֵיךְ תִּתְנַהֵג בְּמִדַּת הָעֲנָוָה לָלֶכֶת בָּהּ תָּמִיד: כָּל דְּבָרֶיךָ יִהְיוּ בְּנַחַת, וְרֹאשְׁךָ כָּפוּף, וְעֵינֶיךָ יַבִּיטוּ לְמַטָּה לָאָרֶץ וְלִבְּךָ לְמַעְלָה, וְאַל תַּבִּיט בִּפְנֵי אָדָם בְּדַבֶּרְךָ עִמּוֹ. וְכָל אָדָם יִהְיֶה גָּדוֹל מִמְּךָ בְּעֵינֶיךָ: אִם חָכָם אוֹ עָשִׁיר הוּא — עָלֶיךָ לְכַבְּדוֹ. וְאִם רָשׁ הוּא, וְאַתָּה עָשִׁיר אוֹ חָכָם מִמֶּנּוּ — חֲשׁוֹב בְּלִבְּךָ כִּי אַתָּה חַיָּב, וְהוּא זַכַּאי. שֶׁאִם הוּא חוֹטֵא — הוּא שׁוֹגֵג, וְאַתָּה מֵזִיד.

בְּכָל דְּבָרֶיךָ וּמַעֲשֶׂיךָ וּמַחְשְׁבוֹתֶיךָ וּבְכָל עֵת — חֲשׁוֹב בְּלִבְּךָ כְּאִלּוּ אַתָּה עוֹמֵד לִפְנֵי הַקָּדוֹשׁ-בָּרוּךְ-הוּא, וּשְׁכִינָתוֹ עָלֶיךָ, כִּי כְּבוֹדוֹ מָלֵא הָעוֹלָם. וּדְבָרֶיךָ יִהְיוּ בְּאֵימָה וּבְיִרְאָה כְּעֶבֶד לִפְנֵי רַבּוֹ, וְתִתְבַּיֵּשׁ מִכָּל אָדָם. וְאִם יִקְרָאֲךָ אִישׁ — אַל תַּעֲנֵהוּ בְּקוֹל רָם, רַק בְּנַחַת — כְּעוֹמֵד לִפְנֵי רַבּוֹ.

וְתִזָּהֵר לִקְרוֹת הַתּוֹרָה תָּמִיד אֲשֶׁר תּוּכַל לְקַיְּמָהּ. וְכַאֲשֶׁר תָּקוּם מִן הַסֵּפֶר — תְּחַפֵּשׂ בַּאֲשֶׁר לָמַדְתָּ אִם יֵשׁ בּוֹ דָּבָר אֲשֶׁר תּוּכַל לְקַיְּמוֹ. וּתְחַפֵּשׂ מַעֲשֶׂיךָ בַּבֹּקֶר וּבָעֶרֶב, וּבָזֶה יִהְיוּ כָּל יָמֶיךָ בִּתְשׁוּבָה.

וְהָסֵר כָּל דִּבְרֵי הָעוֹלָם מִלִּבְּךָ בְּעֵת הַתְּפִלָּה, וְהָכֵן לִבְּךָ לִפְנֵי הַשֵּׁם יִתְבָּרַךְ, וְטַהֵר רַעְיוֹנֶיךָ, וַחֲשׁוֹב הַדִּבּוּר קוֹדֶם שֶׁתּוֹצִיאֵהוּ מִפִּיךָ. וְכֵן תַּעֲשֶׂה כָּל יְמֵי חַיֵּי הֶבְלֶךָ בְּכָל דָּבָר וְדָבָר, וְלֹא תֶחֱטָא, וּבָזֶה יִהְיוּ דְבָרֶיךָ וּמַעֲשֶׂיךָ וּמַחְשְׁבוֹתֶיךָ יְשָׁרִים, וּתְפִלָּתְךָ תְּהֵא זַכָּה וְצַחָה וְכַוָּנָה וּמְקֻבֶּלֶת לִפְנֵי הַמָּקוֹם בָּרוּךְ הוּא, שֶׁנֶּאֱמַר (תהלים י, יז): "תָּכִין לִבָּם תַּקְשִׁיב אָזְנֶךָ".

תִּקְרָא הָאִגֶּרֶת הַזֹּאת פַּעַם אַחַת בְּשָׁבוּעַ, וְלֹא תִפְחֹת — לְקַיְּמָהּ וְלָלֶכֶת בָּהּ תָּמִיד אַחַר הַשֵּׁם יִתְבָּרַךְ, לְמַעַן תַּצְלִיחַ בְּכָל דְּרָכֶיךָ וְתִזְכֶּה לָעוֹלָם הַבָּא הַצָּפוּן לַצַּדִּיקִים. וּבְכָל יוֹם שֶׁתִּקְרָאֶנָּה — יַעֲנוּךָ מִן הַשָּׁמַיִם כַּאֲשֶׁר יַעֲלֶה עַל לִבְּךָ לִשְׁאוֹל עַד עוֹלָם. אָמֵן וְכֵן יְהִי רָצוֹן.

יְהִי רָצוֹן מִלְּפָנֶיךָ יְהֹוָה אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ, שֶׁתּוֹלִיכֵנוּ לְשָׁלוֹם וְתַצְעִידֵנוּ לְשָׁלוֹם וְתַדְרִיכֵנוּ לְשָׁלוֹם, וְתִסְמְכֵנוּ לְשָׁלוֹם, וְתַגִּיעֵנוּ לִמְחוֹז חֶפְצֵנוּ לְחַיִּים וּלְשִׂמְחָה וּלְשָׁלוֹם.

וְתַצִּילֵנוּ מִכַּף כָּל אוֹיֵב וְאוֹרֵב וְלִסְטִים וְחַיּוֹת רָעוֹת בַּדֶּרֶךְ, וּמִכָּל מִינֵי פֻּרְעָנִיּוֹת הַמִּתְרַגְּשׁוֹת לָבוֹא לָעוֹלָם. וְתִשְׁלַח בְּרָכָה בְּכָל מַעֲשֵׂה יָדֵינוּ, וְתִתְּנֵנוּ לְחֵן וּלְחֶסֶד וּלְרַחֲמִים בְּעֵינֶיךָ וּבְעֵינֵי כָל רוֹאֵינוּ.
